       public sealed class PromptComposer
       {
           public string Compose(SessionContext context)

               var stage = context.DeriveStage();

               var snapshot = context.FormatSnapshot();

                   {stageInstructions}

                   ## Game State
                   {snapshot}
                   """;
           }
       }
